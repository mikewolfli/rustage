# The native module that carries the rust_widgets C ABI into ArkTS.
#
# Two things are deliberately *not* copied into this scaffold:
#
#   * `rw_generated.h` — the pinned revision publishes it under `include/`, and a
#     copy here would be a second declaration of the same ABI. Point
#     `RUST_WIDGETS_INCLUDE` at that directory.
#   * `librust_widgets.so` — build it for `aarch64-unknown-linux-ohos` with the
#     `xcomponent` feature and put it under `entry/libs/arm64-v8a/`, or point
#     `RUST_WIDGETS_LIB_DIR` at the directory that holds it. A missing library is a
#     configuration error, not a link that silently omits the bridge.
cmake_minimum_required(VERSION 3.5.0)
project(rustage_entry CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(RUST_WIDGETS_INCLUDE "$ENV{RUST_WIDGETS_INCLUDE}" CACHE PATH
    "The pinned rust-widgets include/ directory that holds rw_generated.h")
set(RUST_WIDGETS_LIB_DIR "$ENV{RUST_WIDGETS_LIB_DIR}" CACHE PATH
    "The directory that holds librust_widgets.so for arm64-v8a")
if(NOT RUST_WIDGETS_LIB_DIR)
    set(RUST_WIDGETS_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/arm64-v8a")
endif()

if(NOT EXISTS "${RUST_WIDGETS_INCLUDE}/rw_generated.h")
    message(FATAL_ERROR
        "RUST_WIDGETS_INCLUDE is unset or does not contain rw_generated.h; set it to the pinned "
        "rust-widgets checkout's include/ directory. Nothing is generated by guessing an ABI.")
endif()

set(RUST_WIDGETS_LIB "${RUST_WIDGETS_LIB_DIR}/librust_widgets.so")
if(NOT EXISTS "${RUST_WIDGETS_LIB}")
    message(FATAL_ERROR
        "${RUST_WIDGETS_LIB} does not exist; build the pinned revision for "
        "aarch64-unknown-linux-ohos with a device profile and the xcomponent feature, as the "
        "README describes, or set RUST_WIDGETS_LIB_DIR to the directory that holds it.")
endif()

add_library(entry SHARED napi_init.cpp)
target_include_directories(entry PRIVATE "${RUST_WIDGETS_INCLUDE}")
target_compile_options(entry PRIVATE -Wall -Wextra)
# `--no-undefined` turns the dangerous case into a configuration failure: a
# `librust_widgets.so` built without a device profile (so without `rw_init` and
# `rw_create_*`) links anyway unless the linker is told to resolve every symbol,
# and the HAP then fails when ArkTS loads the module.
set_target_properties(entry PROPERTIES LINK_FLAGS "-Wl,--no-undefined")
target_link_libraries(entry PUBLIC
    ${RUST_WIDGETS_LIB}
    libace_napi.z.so
    libhilog_ndk.z.so)
